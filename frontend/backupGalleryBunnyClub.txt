import React, { useEffect, useState } from "react";
import axios from "axios";
import { TwitterShareButton, TwitterIcon } from "react-share";
import { Tooltip } from "react-tooltip";
interface Attribute {
  trait_type: string;
  value: string;
}
const GalleryView = () => {
  const [eventsData, setEventsData] = useState([]);

  useEffect(() => {
    const fetchEventsData = async () => {
      try {
        const url =
          "https://sheets.googleapis.com/v4/spreadsheets/1XvXkRTE_4isvVKxYGy8W039SQUUZQM3I7At5ugKnrQc/values/A2:E370?key=AIzaSyAjknT53sMDEOdJ3YtCmjgP3VeSTvSGB3U";
        const response = await axios.get(url);

        const filteredData = response.data.values.filter((item: Array<any>) =>
          item[0].endsWith(".png")
        );
        setEventsData(filteredData);
      } catch (error) {
        console.error("Error fetching data:", error);
      }
    };

    fetchEventsData();
  }, []);

  // Function to parse the attributes string into a list of objects
  const parseAttributes = (attributesString: string): Attribute[] => {
    return attributesString.split(", ").map((attr): Attribute => {
      const [trait_type, value] = attr.split(": ");
      return { trait_type, value };
    });
  };
  const customTooltipStyle: React.CSSProperties = {
    transition: 'opacity 0.3s ease-in-out', // You can customize this style
  };
  return (
    <div className="px-4 py-8 mx-auto max-w-screen-xl grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-6">
      {eventsData.map((item, index) => (
        <div
          key={index}
          className="flex flex-col mb-10 border rounded-lg overflow-hidden shadow-lg transform transition duration-500 hover:scale-105 bg-white "
        >
          <h3 className="text-center text-3xl luckiest mb-4 p-4 bg-hunnys-blue text-white">{`Dolly #${item[1]}`}</h3>
          <img src={item[0]} alt={`Dolly ${item[1]}`} className="w-full h-72 object-cover" />
          <div className="p-4 flex flex-wrap justify-start gap-2">
            {parseAttributes(item[3]).map(
              (attr: Attribute, attrIndex: number) => (
                <span
                  key={attrIndex}
                  className="inline-block rounded-full px-3 py-1 text-sm font-bold text-gray-700 mr-2 mb-2"
                >
                  <span className="uppercase">{`${attr.trait_type}:`}</span>{" "}
                  {`${attr.value}`}
                </span>
              )
            )}
          </div>
          <div className="p-4 flex justify-center">
            <TwitterShareButton
              url={`https://closet.hunnys.io/dolly/${item[1]}`} // Dynamic URL for sharing specific doll
              title={`Check out this Dolly #${item[1]} from closet.hunnys.io / @hunnysNFT!`}
              className="transform transition duration-500 hover:scale-125"
            >
              <img
                src={"/icons/GalleryX.svg"}
                alt={"Hunnys X"}
                width={38}
                height={38}
                className="cursor-pointer xs:hidden md:inline"
              />
            </TwitterShareButton>
          </div>
        </div>
      ))}
    </div>
  );
};

export default GalleryView;
